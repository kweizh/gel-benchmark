import asyncio, gel
from decimal import Decimal
import ledger_api as api

async def main():
    client = gel.create_async_client()
    try:
        await api.create_account(client, code='CONC-A', opening_balance=Decimal('1000.00'))
        await api.create_account(client, code='CONC-B', opening_balance=Decimal('0.00'))

        # 20 concurrent distinct-key transfers of 10.00 each from A to B
        async def do(i):
            return await api.create_transfer(client, sender='CONC-A', recipient='CONC-B', amount=Decimal('10.00'), idempotency_key=f'distinct-{i}')

        results = await asyncio.gather(*(do(i) for i in range(20)))
        assert all(r['created'] for r in results), results
        ids = {r['id'] for r in results}
        assert len(ids) == 20, ids
        balA = await api.get_balance(client, account='CONC-A')
        balB = await api.get_balance(client, account='CONC-B')
        print('after distinct transfers', balA, balB)
        assert balA == '800.00' and balB == '200.00'

        # M concurrent transfers sharing one idempotency key
        async def do_shared():
            return await api.create_transfer(client, sender='CONC-A', recipient='CONC-B', amount=Decimal('5.00'), idempotency_key='shared-key')

        shared_results = await asyncio.gather(*(do_shared() for _ in range(20)))
        created_true = [r for r in shared_results if r['created']]
        created_false = [r for r in shared_results if not r['created']]
        print('created True count:', len(created_true), 'False count:', len(created_false))
        assert len(created_true) == 1
        assert len(created_false) == 19
        ids = {r['id'] for r in shared_results}
        assert len(ids) == 1

        balA2 = await api.get_balance(client, account='CONC-A')
        balB2 = await api.get_balance(client, account='CONC-B')
        print('after shared transfer', balA2, balB2)
        assert balA2 == '795.00' and balB2 == '205.00'

        # Overdraw race: 20 concurrent transfers of 100 each from an account with only 250 balance -> should allow only 2 to succeed (200), rest InsufficientFunds, balance never negative
        await api.create_account(client, code='CONC-C', opening_balance=Decimal('250.00'))
        await api.create_account(client, code='CONC-D', opening_balance=Decimal('0.00'))

        async def do_overdraw(i):
            try:
                r = await api.create_transfer(client, sender='CONC-C', recipient='CONC-D', amount=Decimal('100.00'), idempotency_key=f'overdraw-{i}')
                return ('ok', r)
            except api.InsufficientFunds as e:
