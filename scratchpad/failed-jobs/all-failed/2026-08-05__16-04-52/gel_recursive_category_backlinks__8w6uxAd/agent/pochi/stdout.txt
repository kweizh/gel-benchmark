import gel
client = gel.create_client()

# Check for cycles
result = client.query("""
    WITH MODULE default
    SELECT Category { slug, parent: { slug } }
""")

# Build parent map
parent_map = {}
for cat in result:
    parent_map[cat.slug] = cat.parent.slug if cat.parent else None

# Find cycles
def find_cycle(start):
    visited = set()
    path = []
    current = start
    while current is not None:
        if current in visited:
            # Found a cycle
            cycle_start = path.index(current)
            return path[cycle_start:] + [current]
        visited.add(current)
        path.append(current)
        current = parent_map.get(current)
    return None

cycles = []
for slug in parent_map:
    cycle = find_cycle(slug)
    if cycle and cycle not in cycles:
        # Check if this is a new cycle (not just a different start point)
        is_new = True
        for existing in cycles:
            if set(cycle) == set(existing):
                is_new = False
                break
        if is_new:
            cycles.append(cycle)
            print(f"Cycle: {' -> '.join(cycle)}")

if not cycles:
    print("No cycles found")
